```java
// FileAdapter.java - 添加缩略图加载
public void onBindViewHolder(ViewHolder holder, int position) {
    // ... 现有代码 ...
    
    // 加载视频缩略图
    loadThumbnail(holder.thumbnail, fileItem.getFile().getAbsolutePath());
}

private void loadThumbnail(ImageView imageView, String videoPath) {
    // 使用Glide或手动提取
    new Thread(() -> {
        try {
            MediaMetadataRetriever retriever = new MediaMetadataRetriever();
            retriever.setDataSource(videoPath);
            Bitmap bitmap = retriever.getFrameAtTime(1000000); // 1秒处
            retriever.release();
            
            imageView.post(() -> {
                if (bitmap != null) {
                    imageView.setImageBitmap(bitmap);
                }
            });
        } catch (Exception e) {
            // 使用默认图标
            imageView.post(() -> imageView.setImageResource(R.drawable.ic_video_default));
        }
    }).start();
}
```

**优势**：
- ✅ 视觉冲击力强，易于识别
- ✅ 缩略图帮助快速定位视频
- ✅ 信息展示更清晰
- ✅ 现代化卡片设计

**考虑**：
- 需要添加缩略图缓存机制（避免重复加载）
- 首次加载可能稍慢（可显示加载占位符）

```java
private void showGestureHintIfNeeded() {
    SharedPreferences prefs = getSharedPreferences("app_prefs", MODE_PRIVATE);
    boolean hasShownHint = prefs.getBoolean("gesture_hint_shown", false);
    
    if (!hasShownHint) {
        View hintOverlay = findViewById(R.id.gesture_hint_overlay);
        hintOverlay.setVisibility(View.VISIBLE);
        
        findViewById(R.id.close_hint_button).setOnClickListener(v -> {
            hintOverlay.setVisibility(View.GONE);
            prefs.edit().putBoolean("gesture_hint_shown", true).apply();

        // 5秒后自动关闭
        hintOverlay.postDelayed(() -> {
            hintOverlay.setVisibility(View.GONE);
            prefs.edit().putBoolean("gesture_hint_shown", true).apply();
        }, 5000);
    }
}
```

```java
// HomeFragment.java
private SearchView searchView;
private List<FileItem> searchResults = new ArrayList<>();

private void setupSearchView() {
    searchView = view.findViewById(R.id.search_view);
    searchView.setOnQueryTextListener(new SearchView.OnQueryTextListener() {
        @Override
        public boolean onQueryTextChange(String newText) {
            filterVideosByQuery(newText);
            return true;
        }
        
        @Override
        public boolean onQueryTextSubmit(String query) {
            filterVideosByQuery(query);
            return true;
        }
    });
}

private void filterVideosByQuery(String query) {
    if (query.isEmpty()) {
        // 显示全部
        adapter.updateFiles(displayedFiles);
        return;
    }
    
    searchResults.clear();
    String lowerQuery = query.toLowerCase();
    
    for (FileItem item : displayedFiles) {
        if (item.getName().toLowerCase().contains(lowerQuery)) {
            searchResults.add(item);
        }
    }
    
    adapter.updateFiles(searchResults);
    
    // 显示结果数量
    Toast.makeText(getContext(), 
        "找到 " + searchResults.size() + " 个视频", 
        Toast.LENGTH_SHORT).show();
}
```

```java
// HomeFragment.java
private TextView scanProgressText;
private ProgressBar scanProgressBar;
private int scannedCount = 0;

private void loadFiles() {
    // 显示加载UI
    showLoadingState();
    
    new Thread(() -> {
        scannedCount = 0;
        List<FileItem> files = FileUtils.scanAllFilesWithProgress(
            targetDirectories, 
            currentFilter,
            progress -> {
                // 更新进度
                mainHandler.post(() -> {
                    scannedCount++;
                    scanProgressText.setText(
                        "正在扫描视频... 已找到 " + scannedCount + " 个视频"
                    );
                    scanProgressBar.setProgress(progress);
                });
            }
        );
        
        mainHandler.post(() -> {
            hideLoadingState();
            allFiles = files;
            applyFilterAndSort();
        });
    }).start();
}

private void showLoadingState() {
    skeletonLayout.setVisibility(View.VISIBLE);
    recyclerView.setVisibility(View.GONE);
    emptyView.setVisibility(View.GONE);
}

private void hideLoadingState() {
    skeletonLayout.setVisibility(View.GONE);
    recyclerView.setVisibility(View.VISIBLE);
}
```
